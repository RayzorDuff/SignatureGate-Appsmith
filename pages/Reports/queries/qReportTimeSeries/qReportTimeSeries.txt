WITH raw_params AS (
  SELECT
    lower(NULLIF({{ this.params?.bucket || 'month' }}, ''))::text AS requested_bucket,

    GREATEST(
      1,
      LEAST(
        COALESCE(NULLIF({{ this.params?.periods || '12' }}, '')::int, 12),
        120
      )
    ) AS requested_periods,

    NULLIF({{ this.params?.start_date || '' }}, '')::timestamptz AS start_override,
    NULLIF({{ this.params?.end_date || '' }}, '')::timestamptz AS end_override,

    COALESCE(
      NULLIF({{ this.params?.metric_keys_json || '[]' }}, '')::jsonb,
      '[]'::jsonb
    ) AS metric_keys_json,

    lower({{ appsmith.user.email }})::text AS current_email
),

params AS (
  SELECT
    CASE
      WHEN requested_bucket IN ('day', 'week', 'month', 'year')
        THEN requested_bucket
      ELSE 'month'
    END AS bucket,

    CASE
      WHEN requested_bucket = 'day' THEN interval '1 day'
      WHEN requested_bucket = 'week' THEN interval '1 week'
      WHEN requested_bucket = 'year' THEN interval '1 year'
      ELSE interval '1 month'
    END AS step,

    CASE
      WHEN requested_bucket = 'day' THEN 'YYYY-MM-DD'
      WHEN requested_bucket = 'week' THEN 'IYYY-"W"IW'
      WHEN requested_bucket = 'year' THEN 'YYYY'
      ELSE 'YYYY-MM'
    END AS label_format,

    requested_periods,
    start_override,
    end_override,
    metric_keys_json,
    current_email
  FROM raw_params
),

me AS (
  SELECT person_id, member_id, is_practitioner,
         is_document_reviewer, is_donations_reviewer
  FROM public.issue19_current_release_actor((SELECT current_email FROM params))
  WHERE is_practitioner OR is_document_reviewer OR is_donations_reviewer
),

auth AS (
  SELECT
    p.*,
    me.person_id AS viewer_person_id,
    me.is_practitioner,
    me.is_document_reviewer,
    me.is_donations_reviewer
  FROM params p
  JOIN me ON TRUE
),

bounds AS (
  SELECT
    bucket,
    step,
    label_format,
    metric_keys_json,

    date_trunc(
      bucket,
      COALESCE(
        start_override,
        date_trunc(bucket, now()) - ((requested_periods - 1) * step)
      )
    ) AS start_ts,

    date_trunc(
      bucket,
      COALESCE(end_override, now())
    ) + step AS exclusive_end_ts
  FROM auth
),

buckets AS (
  SELECT
    gs AS bucket_start,
    gs + b.step AS bucket_end,
    b.label_format
  FROM bounds b
  CROSS JOIN LATERAL generate_series(
    b.start_ts,
    b.exclusive_end_ts - b.step,
    b.step
  ) AS gs
),

scoped_members AS (
  SELECT DISTINCT
    m.member_id
  FROM public.member_profiles m
  CROSS JOIN auth a
  WHERE lower(COALESCE(m.status, 'active')) = 'active'
    AND (
      a.is_document_reviewer IS TRUE
      OR (a.is_practitioner AND EXISTS (
        SELECT 1
        FROM public.member_practitioner_assignments assignment
        WHERE assignment.member_id = m.member_id
          AND assignment.practitioner_person_id = a.viewer_person_id
          AND assignment.status = 'active'
      ))
    )
),

scoped_donations AS (
  SELECT d.*, contributor.contributor_type,
         EXISTS (
           SELECT 1 FROM public.members participant
           WHERE participant.person_id = contributor.person_id
             AND participant.status = 'active'
         ) AS has_participant_capacity
  FROM public.donations d
  LEFT JOIN public.contributors contributor
    ON contributor.contributor_id = d.contributor_id
  CROSS JOIN auth a
  WHERE a.is_donations_reviewer IS TRUE
    AND d.donor_kind IN ('identified', 'anonymous', 'unresolved')
),

selected_metrics AS (
  SELECT *
  FROM (
    SELECT 'agreements_signed'::text AS metric_key, 'Agreements signed'::text AS metric_label, 10 AS sort_order
    UNION ALL SELECT 'releases_sent', 'Releases sent', 20
    UNION ALL SELECT 'donations_count', 'Donations received', 30
    UNION ALL SELECT 'donations_dollars', 'Donation dollars', 40
    UNION ALL SELECT 'contributor_donations_count', 'Contributor-linked donations', 45
    UNION ALL SELECT 'contributor_donations_dollars', 'Contributor-linked donation dollars', 46
    UNION ALL SELECT 'participant_donations_count', 'Participant-linked donations (subset)', 50
    UNION ALL SELECT 'participant_donations_dollars', 'Participant-linked donation dollars (subset)', 60
    UNION ALL SELECT 'anonymous_donations_count', 'Anonymous donations', 70
    UNION ALL SELECT 'anonymous_donations_dollars', 'Anonymous donation dollars', 80
    UNION ALL SELECT 'individual_donations_count', 'Individual contributor donations', 90
    UNION ALL SELECT 'individual_donations_dollars', 'Individual contributor donation dollars', 100
    UNION ALL SELECT 'organization_donations_count', 'Company contributor donations', 110
    UNION ALL SELECT 'organization_donations_dollars', 'Company contributor donation dollars', 120
    UNION ALL SELECT 'unresolved_donations_count', 'Unresolved donations', 130
    UNION ALL SELECT 'unresolved_donations_dollars', 'Unresolved donation dollars', 140
    UNION ALL SELECT 'participants_created', 'Participants created', 150
    UNION ALL SELECT 'contributors_created', 'Contributors created', 160
  ) m
  CROSS JOIN bounds b
  CROSS JOIN auth a
  WHERE (
    a.is_donations_reviewer IS TRUE
    OR m.metric_key NOT IN (
      'donations_count',
      'donations_dollars',
      'contributor_donations_count', 'contributor_donations_dollars', 'contributors_created',
      'participant_donations_count',
      'participant_donations_dollars',
      'anonymous_donations_count',
      'anonymous_donations_dollars',
      'individual_donations_count', 'individual_donations_dollars',
      'organization_donations_count', 'organization_donations_dollars',
      'unresolved_donations_count', 'unresolved_donations_dollars'
    )
  )
  AND (
    jsonb_array_length(b.metric_keys_json) = 0
    OR m.metric_key IN (
      -- Accept retained development selections without exposing legacy labels.
      SELECT CASE value
        WHEN 'members_created' THEN 'participants_created'
        WHEN 'member_donations_count' THEN 'participant_donations_count'
        WHEN 'member_donations_dollars' THEN 'participant_donations_dollars'
        ELSE value END
      FROM jsonb_array_elements_text(b.metric_keys_json)
    )
  )
),

agreement_counts AS (
  SELECT
    b.bucket_start,
    'agreements_signed'::text AS metric_key,
    COUNT(ma.member_agreement_id)::numeric AS value
  FROM buckets b
  LEFT JOIN public.member_agreements ma
    ON ma.member_id IN (SELECT member_id FROM scoped_members)
   AND lower(COALESCE(ma.status, '')) IN ('signed', 'complete', 'completed')
   AND COALESCE(ma.signed_at, ma.reviewed_at, ma.updated_at) >= b.bucket_start
   AND COALESCE(ma.signed_at, ma.reviewed_at, ma.updated_at) < b.bucket_end
  GROUP BY b.bucket_start
),

release_counts AS (
  SELECT
    b.bucket_start,
    'releases_sent'::text AS metric_key,
    COUNT(r.release_id)::numeric AS value
  FROM buckets b
  LEFT JOIN public.releases r
    ON r.member_id IN (SELECT member_id FROM scoped_members)
   AND COALESCE(r.released_at, r.created_at) >= b.bucket_start
   AND COALESCE(r.released_at, r.created_at) < b.bucket_end
   AND lower(COALESCE(r.status, 'issued')) <> 'voided'
  GROUP BY b.bucket_start
),

donation_metrics AS (
  SELECT b.bucket_start, category.metric_key,
         CASE WHEN category.is_dollars
           THEN COALESCE(SUM(d.amount_cents), 0)::numeric / 100.0
           ELSE COUNT(d.donation_id)::numeric
         END AS value
  FROM buckets b
  CROSS JOIN (VALUES
    ('donations_count', 'all', false), ('donations_dollars', 'all', true),
    ('contributor_donations_count', 'contributor', false),
    ('contributor_donations_dollars', 'contributor', true),
    ('participant_donations_count', 'participant', false),
    ('participant_donations_dollars', 'participant', true),
    ('anonymous_donations_count', 'anonymous', false),
    ('anonymous_donations_dollars', 'anonymous', true),
    ('individual_donations_count', 'individual', false),
    ('individual_donations_dollars', 'individual', true),
    ('organization_donations_count', 'organization', false),
    ('organization_donations_dollars', 'organization', true),
    ('unresolved_donations_count', 'unresolved', false),
    ('unresolved_donations_dollars', 'unresolved', true)
  ) category(metric_key, kind, is_dollars)
  LEFT JOIN scoped_donations d
    ON COALESCE(d.donated_at, d.created_at) >= b.bucket_start
   AND COALESCE(d.donated_at, d.created_at) < b.bucket_end
   AND lower(COALESCE(d.status, '')) IN ('verified', 'imported', 'pending_review')
   AND (category.kind = 'all'
     OR (category.kind = 'contributor' AND d.donor_kind = 'identified'
         AND d.contributor_id IS NOT NULL AND d.contributor_type IN ('individual', 'organization'))
     OR (category.kind = 'participant' AND d.donor_kind = 'identified'
         AND d.contributor_type = 'individual' AND d.has_participant_capacity)
     OR (category.kind = 'individual' AND d.donor_kind = 'identified'
         AND d.contributor_type = 'individual')
     OR (category.kind = 'organization' AND d.donor_kind = 'identified'
         AND d.contributor_type = 'organization')
     OR (category.kind IN ('anonymous', 'unresolved')
         AND d.donor_kind = category.kind))
  GROUP BY b.bucket_start, category.metric_key, category.is_dollars
),

participant_counts AS (
  SELECT
    b.bucket_start,
    'participants_created'::text AS metric_key,
    COUNT(m.member_id)::numeric AS value
  FROM buckets b
  CROSS JOIN auth a
  LEFT JOIN public.members m
    ON (a.is_document_reviewer IS TRUE OR (a.is_practitioner IS TRUE AND EXISTS (
      SELECT 1 FROM public.member_practitioner_assignments assignment
      WHERE assignment.member_id = m.member_id
        AND assignment.practitioner_person_id = a.viewer_person_id
        AND assignment.status = 'active'
    )))
   AND m.created_at >= b.bucket_start
   AND m.created_at < b.bucket_end
   AND lower(COALESCE(m.status, 'active')) <> 'deleted'
  GROUP BY b.bucket_start
),

-- Capacity creation is distinct from person/company creation. Later archive/merge
-- or an ended enrollment must not erase a historical creation event.
contributor_counts AS (
  SELECT b.bucket_start, 'contributors_created'::text AS metric_key,
         COUNT(c.contributor_id)::numeric AS value
  FROM buckets b
  CROSS JOIN auth a
  LEFT JOIN public.contributors c
    ON a.is_donations_reviewer IS TRUE
   AND c.created_at >= b.bucket_start
   AND c.created_at < b.bucket_end
  GROUP BY b.bucket_start
),

all_metric_values AS (
  SELECT * FROM agreement_counts
  UNION ALL SELECT * FROM release_counts
  UNION ALL SELECT * FROM donation_metrics
  UNION ALL SELECT * FROM participant_counts
  UNION ALL SELECT * FROM contributor_counts
)

SELECT
  mv.bucket_start,
  to_char(mv.bucket_start, b.label_format) AS bucket_label,
  mv.metric_key,
  sm.metric_label,
  sm.sort_order,
  mv.value
FROM all_metric_values mv
JOIN selected_metrics sm
  ON sm.metric_key = mv.metric_key
CROSS JOIN bounds b
ORDER BY mv.bucket_start, sm.sort_order;
