WITH member_change AS (
  UPDATE public.member_emails e
  SET is_verified = true, verified_at = now(),
      verified_by = NULL, verification_source = 'individual_profile',
      verification_notes = NULLIF({{ inpIndividualRoleContactReason.text || '' }}, ''),
      updated_at = now()
  WHERE {{ selIndividualContactCapacity.selectedOptionValue || '' }} = 'member'
    AND e.member_email_id = {{ selIndividualRoleContactToArchive.selectedOptionValue }}::uuid
    AND e.status = 'active'
    AND e.member_id = (
      SELECT m.member_id FROM public.members m
      WHERE m.person_id = {{ qIndividualProfile.data?.[0]?.party_id || null }}::uuid
        AND m.status = 'active'
    )
    AND public.issue19_has_role({{ appsmith.user.email }}, 'document_reviewer')
  RETURNING e.member_email_id AS contact_id
),
contributor_change AS (
  UPDATE public.contributor_emails e
  SET is_verified = true
  WHERE {{ selIndividualContactCapacity.selectedOptionValue || '' }} = 'contributor'
    AND e.contributor_email_id = {{ selIndividualRoleContactToArchive.selectedOptionValue }}::uuid
    AND e.status = 'active'
    AND e.contributor_id = (
      SELECT c.contributor_id FROM public.contributors c
      WHERE c.person_id = {{ qIndividualProfile.data?.[0]?.party_id || null }}::uuid
        AND c.status = 'active' AND c.contributor_type = 'individual'
    )
    AND public.issue19_has_role({{ appsmith.user.email }}, 'donations_reviewer')
  RETURNING e.contributor_email_id AS contact_id
)
SELECT COALESCE((SELECT contact_id FROM member_change),
                (SELECT contact_id FROM contributor_change))::text AS contact_id;
