WITH member_target AS (
  SELECT e.member_email_id,e.member_id
  FROM public.member_emails e JOIN public.members m ON m.member_id=e.member_id
  WHERE {{ selIndividualContactCapacity.selectedOptionValue || '' }}='member'
    AND e.member_email_id={{ selIndividualRoleContactToArchive.selectedOptionValue || null }}::uuid
    AND e.status='active' AND m.status='active'
    AND m.person_id={{ qIndividualProfile.data?.[0]?.party_id || null }}::uuid
    AND public.issue19_has_role({{ appsmith.user.email }},'document_reviewer')
), member_cleared AS (
  UPDATE public.member_emails e SET is_primary=false,updated_at=now()
  WHERE e.member_id=(SELECT member_id FROM member_target)
    AND e.member_email_id<>(SELECT member_email_id FROM member_target)
    AND e.status='active' AND e.is_primary=true
  RETURNING e.member_email_id
), member_change AS (
  UPDATE public.member_emails e SET is_primary=true,updated_at=now()
  WHERE e.member_email_id=(SELECT member_email_id FROM member_target)
    AND (SELECT count(*) FROM member_cleared)>=0
  RETURNING e.member_email_id AS contact_id
), contributor_target AS (
  SELECT e.contributor_email_id,e.contributor_id
  FROM public.contributor_emails e
  JOIN public.contributors c ON c.contributor_id=e.contributor_id
  WHERE {{ selIndividualContactCapacity.selectedOptionValue || '' }}='contributor'
    AND e.contributor_email_id={{ selIndividualRoleContactToArchive.selectedOptionValue || null }}::uuid
    AND e.status='active' AND c.status='active'
    AND c.contributor_type='individual'
    AND c.person_id={{ qIndividualProfile.data?.[0]?.party_id || null }}::uuid
    AND public.issue19_has_role({{ appsmith.user.email }},'donations_reviewer')
), contributor_cleared AS (
  UPDATE public.contributor_emails e SET is_primary=false,updated_at=now()
  WHERE e.contributor_id=(SELECT contributor_id FROM contributor_target)
    AND e.contributor_email_id<>(SELECT contributor_email_id FROM contributor_target)
    AND e.status='active' AND e.is_primary=true
  RETURNING e.contributor_email_id
), contributor_change AS (
  UPDATE public.contributor_emails e SET is_primary=true,updated_at=now()
  WHERE e.contributor_email_id=(SELECT contributor_email_id FROM contributor_target)
    AND (SELECT count(*) FROM contributor_cleared)>=0
  RETURNING e.contributor_email_id AS contact_id
)
SELECT COALESCE((SELECT contact_id FROM member_change),(SELECT contact_id FROM contributor_change))::text AS contact_id;
