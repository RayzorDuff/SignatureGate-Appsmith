SELECT cc.contact_id::text AS contact_id, cc.contact_kind,cc.contact_value,cc.is_primary,cc.is_verified,
e.mailing_subscription_status,e.listmonk_sync_status
FROM public.issue19_contributor_contacts({{ appsmith.user.email }},'individual',{{ qIndividualProfile.data?.[0]?.party_id || null }}::uuid) cc
LEFT JOIN public.contributor_emails e ON e.contributor_email_id=cc.contact_id AND cc.contact_kind='email'
ORDER BY cc.contact_kind,cc.is_primary DESC,cc.contact_value;
