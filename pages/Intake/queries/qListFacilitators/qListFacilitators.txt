SELECT r.person_id,p.first_name,p.last_name,p.display_name,COALESCE(a.email,'') AS email FROM public.person_roles r JOIN public.people p ON p.person_id=r.person_id LEFT JOIN public.person_app_accounts a ON a.person_id=r.person_id AND a.status='active' WHERE r.role_key='practitioner' ORDER BY p.last_name,p.first_name,p.display_name;
