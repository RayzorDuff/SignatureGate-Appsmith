WITH me AS (
  SELECT
    member_id,
    public.can_view_donation_financials(member_id) AS can_view_donation_financials
  FROM public.member_profiles
  WHERE lower(email) = lower({{ appsmith.user.email }})
    AND is_facilitator IS TRUE
    AND status = 'active'
  LIMIT 1
)
SELECT d.*
FROM public.donations d
CROSS JOIN me
WHERE me.can_view_donation_financials
  AND d.member_id = {{ appsmith.store.member_id }}::uuid
  AND (
    {{ Boolean(this.params.show_rejected) }}
    OR lower(COALESCE(d.status, '')) <> 'rejected'
  )
ORDER BY
  d.donated_at DESC NULLS LAST,
  d.created_at DESC;
