WITH me AS (
  SELECT
    member_id,
    COALESCE(is_document_reviewer, FALSE) AS is_document_reviewer
  FROM public.member_profiles
  WHERE lower(email) = lower({{ appsmith.user.email }})
    AND status = 'active'
  LIMIT 1
)
SELECT
  ma.member_agreement_id,
  ma.created_at,
  ma.status,
  ma.signature_method,
  ma.member_signed_at,
  ma.facilitator_signed_at,
  ma.opensign_document_id,
  COALESCE(at.name, 'Paper Agreement - No Template') AS template_name,
  ma.facilitator_id,
  ma.evidence,
  ma.review_notes,
	ma.documenso_document_id,
	ma.documenso_external_id,
	ma.canceled_at,
	ma.canceled_by,
	ma.cancel_reason,
	ma.expired_at	
FROM public.member_agreements ma
LEFT JOIN public.agreement_templates at
  ON at.agreement_template_id = ma.agreement_template_id
CROSS JOIN me
WHERE ma.member_id = {{ appsmith.store.member_id }}::uuid
  AND (
    me.is_document_reviewer = TRUE
    OR EXISTS (
      SELECT 1
      FROM public.member_facilitators mf
      WHERE mf.member_id = ma.member_id
        AND mf.facilitator_id = me.member_id
        AND mf.status = 'active'
    )
  )
ORDER BY ma.created_at DESC;
