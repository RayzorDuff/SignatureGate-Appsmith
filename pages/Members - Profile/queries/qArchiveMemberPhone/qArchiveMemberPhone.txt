UPDATE public.member_phones
SET
  status = 'archived',
  archived_at = now(),
  archived_by = {{ appsmith.store.facilitator_id }}::uuid,
  archive_reason = {{ this.params.reason || 'Archived from Members - Profile' }},
  updated_at = now()

RETURNING *;
