WITH actor AS (SELECT * FROM public.application_access({{ appsmith.user.email }})),
contributors AS (
 SELECT 'contributor:'||c.contributor_id::text AS selection_value,c.contributor_id,m.member_id,c.contributor_type,c.display_name,c.first_name,c.last_name,NULL::text AS email,NULL::text AS phone,
 c.display_name||CASE WHEN m.member_id IS NOT NULL THEN ' (Contributor and Participant)' WHEN c.contributor_type='organization' THEN ' (Company Contributor)' ELSE ' (Individual Contributor)' END AS selection_label
 FROM public.contributor_profiles c CROSS JOIN actor a
 LEFT JOIN LATERAL(SELECT member_id FROM public.members WHERE person_id=c.person_id AND status='active' ORDER BY created_at DESC LIMIT 1) m ON true
 WHERE c.status='active' AND (a.is_donations_reviewer OR public.application_member_access({{ appsmith.user.email }},m.member_id,true))
), participants AS (
 SELECT 'member:'||m.member_id::text,NULL::uuid,m.member_id,'individual'::text,
 concat_ws(' ',m.first_name,m.last_name),m.first_name,m.last_name,m.email,m.phone,
 concat_ws(' ',m.first_name,m.last_name)||' (Participant; creates Contributor capacity)'
 FROM public.member_profiles m CROSS JOIN actor a
 WHERE m.status='active' AND (a.is_donations_reviewer OR public.application_member_access({{ appsmith.user.email }},m.member_id,true))
 AND NOT EXISTS(SELECT 1 FROM public.contributors c WHERE c.person_id=m.person_id AND c.status='active')
)
SELECT * FROM contributors UNION ALL SELECT * FROM participants ORDER BY display_name,selection_value LIMIT 750;
