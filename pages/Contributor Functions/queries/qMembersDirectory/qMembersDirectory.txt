WITH me AS (
  SELECT
    member_id,
    COALESCE(is_donations_reviewer, FALSE) AS is_donations_reviewer
  FROM public.member_profiles
  WHERE lower(email) = lower({{ appsmith.user.email }})
    AND status = 'active'
  LIMIT 1
),
contributor_rows AS (
  SELECT
    'contributor:' || c.contributor_id::text AS selection_value,
    c.contributor_id,
    cml.member_id,
    c.contributor_type,
    c.display_name,
    c.first_name,
    c.last_name,
    ce.email,
    cp.phone,
    CASE
      WHEN c.contributor_type = 'organization'
        THEN c.display_name || ' (organization contributor)'
      WHEN cml.member_id IS NOT NULL
        THEN c.display_name || ' (member)'
      ELSE c.display_name || ' (individual contributor)'
    END AS selection_label
  FROM public.contributor_profiles c
  CROSS JOIN me
  LEFT JOIN public.contributor_member_links cml
    ON cml.contributor_id = c.contributor_id
   AND cml.status = 'active'
  LEFT JOIN LATERAL (
    SELECT x.email
    FROM public.contributor_emails x
    WHERE x.contributor_id = c.contributor_id
      AND x.status = 'active'
    ORDER BY x.is_primary DESC, x.created_at
    LIMIT 1
  ) ce ON TRUE
  LEFT JOIN LATERAL (
    SELECT x.phone
    FROM public.contributor_phones x
    WHERE x.contributor_id = c.contributor_id
      AND x.status = 'active'
    ORDER BY x.is_primary DESC, x.created_at
    LIMIT 1
  ) cp ON TRUE
  WHERE c.status = 'active'
    AND (
      me.is_donations_reviewer IS TRUE
      OR (
        cml.member_id IS NOT NULL
        AND EXISTS (
          SELECT 1
          FROM public.member_facilitators mf
          WHERE mf.member_id = cml.member_id
            AND mf.facilitator_id = me.member_id
            AND mf.status = 'active'
        )
      )
    )
),
unlinked_member_rows AS (
  SELECT
    'member:' || m.member_id::text AS selection_value,
    NULL::uuid AS contributor_id,
    m.member_id,
    'individual'::text AS contributor_type,
    COALESCE(
      NULLIF(btrim(concat_ws(' ', m.first_name, m.last_name)), ''),
      m.email,
      'Member ' || m.member_id::text
    ) AS display_name,
    m.first_name,
    m.last_name,
    m.email,
    m.phone,
    COALESCE(
      NULLIF(btrim(concat_ws(' ', m.first_name, m.last_name)), ''),
      m.email,
      'Member ' || m.member_id::text
    ) || ' (member)' AS selection_label
  FROM public.member_profiles m
  CROSS JOIN me
  WHERE m.status = 'active'
    -- A member who already has an individual contributor record should
    -- appear through the contributor row above, even if the contributor/member
    -- link has not yet been established. Selecting the member row would call
    -- ensure_member_contributor() and can collide with that existing contributor.
    AND NOT EXISTS (
      SELECT 1
      FROM public.contributors existing_c
      WHERE existing_c.person_id = m.person_id
        AND existing_c.status = 'active'
    )
    AND NOT EXISTS (
      SELECT 1
      FROM public.contributor_member_links cml
      WHERE cml.member_id = m.member_id
        AND cml.status = 'active'
    )
    AND (
      me.is_donations_reviewer IS TRUE
      OR EXISTS (
        SELECT 1
        FROM public.member_facilitators mf
        WHERE mf.member_id = m.member_id
          AND mf.facilitator_id = me.member_id
          AND mf.status = 'active'
      )
    )
)
SELECT * FROM contributor_rows
UNION ALL
SELECT * FROM unlinked_member_rows
ORDER BY display_name, selection_value
LIMIT 750;
