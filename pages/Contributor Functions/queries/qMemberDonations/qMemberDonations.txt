WITH me AS (
  SELECT
    member_id,
    COALESCE(is_donations_reviewer, FALSE) AS is_donations_reviewer
  FROM public.member_profiles
  WHERE lower(email) = lower({{ appsmith.user.email }})
    AND is_facilitator IS TRUE
    AND status = 'active'
  LIMIT 1
)
SELECT
  d.*,
  c.display_name AS contributor_name,
  c.contributor_type,
  CASE
    WHEN d.donor_kind = 'anonymous' THEN 'Anonymous'
    WHEN d.donor_kind = 'unresolved' THEN 'Unresolved'
    ELSE COALESCE(c.display_name, 'Contributor identity missing')
  END AS donor_display_name
FROM public.donations d
LEFT JOIN public.contributor_profiles c
  ON c.contributor_id = d.contributor_id
CROSS JOIN me
WHERE (
    me.is_donations_reviewer
    OR (
      d.member_id IS NOT NULL
      AND EXISTS (
        SELECT 1
        FROM public.member_facilitators mf
        WHERE mf.member_id = d.member_id
          AND mf.facilitator_id = me.member_id
          AND mf.status = 'active'
      )
    )
  )
  AND (
    CASE
      WHEN left(COALESCE({{ this.params.selection_value || "" }}, ''), 12) = 'contributor:'
        THEN d.contributor_id = substring({{ this.params.selection_value || "" }} from 13)::uuid
      WHEN left(COALESCE({{ this.params.selection_value || "" }}, ''), 7) = 'member:'
        THEN d.member_id = substring({{ this.params.selection_value || "" }} from 8)::uuid
      WHEN {{ this.params.selection_value || "" }} = '__anonymous__'
        THEN COALESCE(d.donor_kind, '') = 'anonymous'
      ELSE FALSE
    END
  )
  AND (
    {{ Boolean(this.params.show_rejected) }}
    OR lower(COALESCE(d.status, '')) <> 'rejected'
  )
ORDER BY
  d.donated_at DESC NULLS LAST,
  d.created_at DESC;
