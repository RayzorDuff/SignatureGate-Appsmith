INSERT INTO public.member_facilitators (
  member_id,
  facilitator_id,
  assigned_by_member_id,
  status,
  notes
)
SELECT
  {{ this.params.member_id }}::uuid,
  {{ this.params.facilitator_id }}::uuid,
  {{ this.params.facilitator_id }}::uuid,
  'active',
  {{ this.params.notes || null }}

WHERE public.application_require_permission({{ appsmith.user.email }},'intake.participant',{{ this.params.member_id || appsmith.store.member_id || null }}::uuid)

ON CONFLICT (member_id, facilitator_id)
DO UPDATE SET
  status = 'active',
  updated_at = now(),
  assigned_by_member_id = EXCLUDED.assigned_by_member_id,
  notes = EXCLUDED.notes
RETURNING member_facilitator_id, member_id, facilitator_id, status;
