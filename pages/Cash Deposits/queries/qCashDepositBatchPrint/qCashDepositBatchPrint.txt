WITH access AS (
 SELECT public.issue19_has_role({{ appsmith.user.email }},'donations_reviewer') AS allowed
)
SELECT *
FROM public.cash_deposit_batch_print p
CROSS JOIN access a
WHERE a.allowed
  AND p.deposit_batch_id = {{ this.params.deposit_batch_id }}::uuid
ORDER BY p.donated_at NULLS LAST, p.donation_id;
