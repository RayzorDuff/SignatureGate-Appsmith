WITH access AS (
  SELECT public.can_view_donation_financials(mp.member_id) AS allowed
  FROM public.member_profiles mp
  WHERE lower(mp.email) = lower({{ appsmith.user.email }})
    AND mp.status = 'active'
  LIMIT 1
)
SELECT
  d.donation_id,
  d.created_at,
  d.donated_at,
  d.amount_cents,
  d.currency,
  d.donor_kind,
  d.member_id,
  d.contributor_id,
  CASE
    WHEN d.donor_kind = 'anonymous' THEN 'Anonymous'
    ELSE COALESCE(c.display_name, d.member_id::text)
  END AS donor_name,
  c.contributor_type,
  d.status AS donation_status,
  CASE
    WHEN e.exclusion_id IS NOT NULL THEN 'deposit_reconciliation_excluded'
    WHEN b.status = 'confirmed' THEN 'deposit_confirmed'
    WHEN b.status = 'prepared' THEN 'deposit_prepared'
    WHEN b.status = 'draft' THEN 'assigned_to_deposit_batch'
    WHEN d.status = 'pending_review' THEN 'pending_review'
    WHEN d.status = 'verified' THEN 'cash_on_hand'
    ELSE d.status
  END AS operational_status,
  b.deposit_batch_id,
  b.status AS batch_status,
  e.exclusion_id,
  e.excluded_at,
  e.excluded_by,
  e.reason AS exclusion_reason,
  e.notes AS exclusion_notes,
  d.notes,
  d.review_notes
FROM public.donations d
LEFT JOIN public.contributor_profiles c
  ON c.contributor_id = d.contributor_id
LEFT JOIN LATERAL (
  SELECT
    b.deposit_batch_id,
    b.status
  FROM public.cash_deposit_batch_items i
  JOIN public.cash_deposit_batches b
    ON b.deposit_batch_id = i.deposit_batch_id
  WHERE i.donation_id = d.donation_id
    AND i.removed_at IS NULL
  ORDER BY i.created_at DESC
  LIMIT 1
) b ON TRUE
LEFT JOIN public.cash_deposit_donation_exclusions e
  ON e.donation_id = d.donation_id
CROSS JOIN access a
WHERE a.allowed
  AND d.provider = 'cash'
  AND d.donor_kind IN ('identified', 'anonymous')
  AND (
    '{{ selCashDepositStatus.selectedOptionValue || "ALL" }}' = 'ALL'
    OR
    (
      CASE
        WHEN e.exclusion_id IS NOT NULL THEN 'deposit_reconciliation_excluded'
        WHEN b.status = 'confirmed' THEN 'deposit_confirmed'
        WHEN b.status = 'prepared' THEN 'deposit_prepared'
        WHEN b.status = 'draft' THEN 'assigned_to_deposit_batch'
        WHEN d.status = 'pending_review' THEN 'pending_review'
        WHEN d.status = 'verified' THEN 'cash_on_hand'
        ELSE d.status
      END
    ) = '{{ selCashDepositStatus.selectedOptionValue }}'
  )
  AND (
    {{ appsmith.store.cash_deposit_hide_ineligible !== false }}
    AND (
      d.status = 'pending_review'
      OR (
        d.status = 'verified'
        AND b.deposit_batch_id IS NULL
        AND e.exclusion_id IS NULL
      )
    )
    OR
    {{ appsmith.store.cash_deposit_hide_ineligible === false }}
  )
ORDER BY d.donated_at DESC NULLS LAST, d.created_at DESC, d.donation_id;
