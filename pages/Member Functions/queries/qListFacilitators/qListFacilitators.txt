SELECT available.practitioner_person_id AS member_id,available.first_name,available.last_name,available.account_email AS email
FROM public.issue19_release_practitioners({{ appsmith.user.email }},NULLIF({{ appsmith.store.member_id || "" }},'')::uuid) available
UNION SELECT p.person_id,p.first_name,p.last_name,a.email FROM public.person_roles role JOIN public.people p USING(person_id)
LEFT JOIN public.person_app_accounts a ON a.person_id=p.person_id AND a.status='active'
WHERE role.role_key='practitioner' AND public.issue19_has_role({{ appsmith.user.email }},'document_reviewer');
