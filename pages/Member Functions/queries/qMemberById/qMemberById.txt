WITH actor AS (
  SELECT member_id
  FROM public.member_profiles
  WHERE lower(email)=lower({{ appsmith.user.email }})
    AND status='active'
  LIMIT 1
)
SELECT m.*
FROM public.member_profiles m
WHERE m.member_id = {{ appsmith.store.member_id || null }}::uuid
  AND (
    public.issue19_has_role({{ appsmith.user.email }}, 'document_reviewer')
    OR EXISTS (
      SELECT 1 FROM public.member_facilitators mf
      JOIN actor ON actor.member_id=mf.facilitator_id
      WHERE mf.member_id=m.member_id AND mf.status='active'
    )
  );
