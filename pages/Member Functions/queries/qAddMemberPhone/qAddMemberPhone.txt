INSERT INTO public.member_phones (
  member_id,
  phone,
  is_primary,
  source,
  notes
)
SELECT
  {{ appsmith.store.member_id }}::uuid,
  NULLIF(btrim({{ this.params.phone }}), ''),
  COALESCE({{ this.params.is_primary }}, false),
  'manual',
  {{ this.params.notes || null }}
WHERE NULLIF(public.normalize_us_phone({{ this.params.phone }}), '') IS NOT NULL
  AND NOT EXISTS (
    SELECT 1
    FROM public.member_phones mp
    WHERE mp.member_id = {{ appsmith.store.member_id }}::uuid
      AND mp.phone_normalized = public.normalize_us_phone({{ this.params.phone }})
      AND COALESCE(mp.status, 'active') = 'active'
  )
RETURNING *;
