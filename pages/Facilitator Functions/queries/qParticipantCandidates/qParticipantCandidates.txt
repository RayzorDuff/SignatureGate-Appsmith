SELECT m.member_id,m.person_id,m.first_name,m.last_name,m.email FROM public.member_profiles m WHERE m.status='active'
AND public.issue19_has_role({{ appsmith.user.email }},'document_reviewer')
AND NOT EXISTS (SELECT 1 FROM public.member_practitioner_assignments a JOIN public.members f ON f.person_id=a.practitioner_person_id
WHERE a.member_id=m.member_id AND a.status='active' AND f.member_id={{ appsmith.store.selected_facilitator_id || null }}::uuid)
ORDER BY m.last_name,m.first_name;
