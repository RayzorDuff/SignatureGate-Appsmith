SELECT a.member_practitioner_assignment_id,a.member_id,m.person_id AS participant_person_id,m.first_name,m.last_name,m.email,a.status,a.created_at,a.updated_at,a.notes,a.practitioner_person_id AS practitioner_member_id
FROM public.member_practitioner_assignments a JOIN public.member_profiles m ON m.member_id=a.member_id
WHERE a.practitioner_person_id={{ appsmith.store.selected_facilitator_id || null }}::uuid
AND public.issue19_has_role({{ appsmith.user.email }},'document_reviewer')
ORDER BY CASE WHEN a.status='active' THEN 0 ELSE 1 END,m.last_name,m.first_name;
