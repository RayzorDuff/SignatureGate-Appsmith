INSERT INTO public.audit_log (actor, action, entity_type, entity_id, details)
SELECT {{ appsmith.user.email }}::text,
       {{ this.params.action }}::text, {{ this.params.entity_type }}::text,
       COALESCE(NULLIF({{ this.params.entity_id || "" }}::text,''),'unknown'),
       COALESCE({{ this.params.details_json || "{}" }}::jsonb,'{}'::jsonb)
WHERE public.application_require_permission({{ appsmith.user.email }}::text,'practitioners.manage')
RETURNING audit_log_id;
